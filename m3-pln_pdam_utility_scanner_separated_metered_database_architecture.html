Tri-Layered Separated Database Architecture

Fully isolated schema separating Employee Profiles (tbl_employees), Auth Passwords (tbl_passwords), and Metered Utility Readings (tbl_meter_readings).

DB 1: tbl_employees Public Profiles

Contains officer roster, departments, and active employment roles.

PRIMARY KEY: emp_id
- full_name, email
- department, utility_type
FK target for auth & audit logs
DB 2: tbl_passwords Restricted Vault

Strictly isolated database storing salts, SHA256 hashes, and security logs.

FOREIGN KEY: emp_id -> tbl_employees
- password_hash (SHA-256)
- salt, failed_attempts
Zero access from metered services
DB 3: tbl_meter_readings Metered Data Storage

Isolated transactional database for field scan entries, consumption, and GPS telemetry.

PRIMARY KEY: reading_id
FOREIGN KEY: officer_id -> tbl_employees
- customer_id, stand_reading, delta
- est_cost, validation_status, anomaly_flag

import androidx.room.Entity

/**
 * Isolated Room Database Entity for Metered Readings.
 * Foreign key officerId links to Employee entity without touching sensitive auth credentials.
 */
@Entity(
    tableName = "tbl_meter_readings",
    foreignKeys = [
        ForeignKey(
            entity = EmployeeEntity::class,
            parentColumns = ["emp_id"],
            childColumns = ["officer_id"],
            onDelete = ForeignKey.RESTRICT
        )
    ],
    indices = [Index(value = ["officer_id"]), Index(value = ["customer_id"])]
)
data class MeterReadingEntity(
    @PrimaryKey val readingId: String,
    val timestamp: Long,
    val utilityType: String, // "PLN" or "PDAM"
    val customerId: String,
    val standReading: String,
    val previousReading: Double,
    val usageDelta: Double,
    val estimatedCost: Double,
    val officerId: String,
    val validationStatus: String, // "VALID" or "ANOMALY"
    val gpsLocation: String,
    val imageFilePath: String? = null
)

@Dao
interface MeterReadingDao {

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertReading(reading: MeterReadingEntity)

    @Query("SELECT * FROM tbl_meter_readings ORDER BY timestamp DESC")
    fun getAllReadings(): Flow<List<MeterReadingEntity>>

    @Query("SELECT * FROM tbl_meter_readings WHERE utilityType = :type ORDER BY timestamp DESC")
    fun getReadingsByUtility(type: String): Flow<List<MeterReadingEntity>>

    suspend fun getAnomalyCount(): Int
}
package com.utility.meterocr.domain.ocr

import androidx.camera.core.ImageAnalysis
import androidx.camera.core.ImageProxy
import com.google.mlkit.vision.common.InputImage
import com.google.mlkit.vision.text.TextRecognition
import com.google.mlkit.vision.text.latin.TextRecognizerOptions

enum class ScanStep { STEP_1_CUSTOMER_ID, STEP_2_STAND_READING }

data class MeterOCRResult(
    val step: ScanStep,
    val extractedValue: String = "",
    val confidence: Float = 0.96f
)

/**
 * 2-Step Offline OCR Analyzer for PLN/PDAM utility meters.
 */
class MeterImageAnalyzer(
    private var currentStep: ScanStep = ScanStep.STEP_1_CUSTOMER_ID,
    private val onStepExtracted: (MeterOCRResult) -> Unit
) : ImageAnalysis.Analyzer {

    private val recognizer = TextRecognition.getClient(TextRecognizerOptions.DEFAULT_OPTIONS)

    fun setScanStep(step: ScanStep) {
        this.currentStep = step
    }

    @androidx.annotation.OptIn(androidx.camera.core.ExperimentalGetImage::class)
    override fun analyze(imageProxy: ImageProxy) {
        val mediaImage = imageProxy.image
        if (mediaImage != null) {
            val inputImage = InputImage.fromMediaImage(mediaImage, imageProxy.imageInfo.rotationDegrees)

            recognizer.process(inputImage)
                .addOnSuccessListener { visionText ->
                    val rawText = visionText.text
                    
                    when (currentStep) {
                        ScanStep.STEP_1_CUSTOMER_ID -> {
                            val customerId = extractCustomerId(rawText)
                            if (customerId.isNotEmpty()) {
                                onStepExtracted(MeterOCRResult(ScanStep.STEP_1_CUSTOMER_ID, customerId))
                            }
                        }
                        ScanStep.STEP_2_STAND_READING -> {
                            val standDigits = extractStandDigits(rawText)
                            if (standDigits.isNotEmpty()) {
                                onStepExtracted(MeterOCRResult(ScanStep.STEP_2_STAND_READING, standDigits))
                            }
                        }
                    }
                }
                .addOnCompleteListener {
                    imageProxy.close()
                }
        } else {
            imageProxy.close()
        }
    }

    private fun extractCustomerId(text: String): String {
        val pattern = Regex("\\b\\d{11,12}\\b")
        return pattern.find(text)?.value ?: ""
    }

    private fun extractStandDigits(text: String): String {
        val pattern = Regex("\\b\\d{4,7}\\b")
        return pattern.find(text)?.value ?: ""
    }
}
