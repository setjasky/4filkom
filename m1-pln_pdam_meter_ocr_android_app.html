package com.utility.meterocr.domain.ocr

import androidx.camera.core.ImageAnalysis
import androidx.camera.core.ImageProxy
import com.google.mlkit.vision.common.InputImage
import com.google.mlkit.vision.text.TextRecognition
import com.google.mlkit.vision.text.latin.TextRecognizerOptions

enum class ScanStep { STEP_1_CUSTOMER_ID, STEP_2_STAND_READING }

data class MeterOCRResult(
    val step: ScanStep,
    val extractedValue: String = "",
    val confidence: Float = 0.96f
)

/**
 * 2-Step Offline OCR Analyzer for PLN/PDAM utility meters.
 * Step 1: Detects and validates 11-12 digit Customer/Meter Serial IDs.
 * Step 2: Detects and validates 4-7 digit Stand Readings (kWh/m³).
 */
class MeterImageAnalyzer(
    private var currentStep: ScanStep = ScanStep.STEP_1_CUSTOMER_ID,
    private val onStepExtracted: (MeterOCRResult) -> Unit
) : ImageAnalysis.Analyzer {

    private val recognizer = TextRecognition.getClient(TextRecognizerOptions.DEFAULT_OPTIONS)

    fun setScanStep(step: ScanStep) {
        this.currentStep = step
    }

    @androidx.annotation.OptIn(androidx.camera.core.ExperimentalGetImage::class)
    override fun analyze(imageProxy: ImageProxy) {
        val mediaImage = imageProxy.image
        if (mediaImage != null) {
            val inputImage = InputImage.fromMediaImage(mediaImage, imageProxy.imageInfo.rotationDegrees)

            recognizer.process(inputImage)
                .addOnSuccessListener { visionText ->
                    val rawText = visionText.text
                    
                    when (currentStep) {
                        ScanStep.STEP_1_CUSTOMER_ID -> {
                            // Extract 11-12 digit PLN/PDAM ID (e.g., 541100234182)
                            val customerId = extractCustomerId(rawText)
                            if (customerId.isNotEmpty()) {
                                onStepExtracted(MeterOCRResult(ScanStep.STEP_1_CUSTOMER_ID, customerId))
                            }
                        }
                        ScanStep.STEP_2_STAND_READING -> {
                            // Extract 4-7 digit Stand Reading (e.g., 014925)
                            val standDigits = extractStandDigits(rawText)
                            if (standDigits.isNotEmpty()) {
                                onStepExtracted(MeterOCRResult(ScanStep.STEP_2_STAND_READING, standDigits))
                            }
                        }
                    }
                }
                .addOnCompleteListener {
                    imageProxy.close()
                }
        } else {
            imageProxy.close()
        }
    }

    private fun extractCustomerId(text: String): String {
        val pattern = Regex("\\b\\d{11,12}\\b")
        return pattern.find(text)?.value ?: ""
    }

    private fun extractStandDigits(text: String): String {
        val pattern = Regex("\\b\\d{4,7}\\b")
        return pattern.find(text)?.value ?: ""
    }
}
package com.utility.meterocr.ui.viewmodel

import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow

enum class MeterType { PLN_ELECTRICITY, PDAM_WATER }

data class OCRScanUiState(
    val selectedType: MeterType = MeterType.PLN_ELECTRICITY,
    val customerId: String = "541100234182",
    val detectedDigits: String = "014925",
    val confidenceScore: Float = 0.98f,
    val previousReading: Double = 14680.0,
    val tariffRatePerUnit: Double = 1444.70,
    val usageDelta: Double = 245.0,
    val estimatedBill: Double = 353951.0,
    val isAnomalyDetected: Boolean = false
)

class MeterViewModel : ViewModel() {

    private val _uiState = MutableStateFlow(OCRScanUiState())
    val uiState: StateFlow<OCRScanUiState> = _uiState.asStateFlow()

    fun updateMeterType(type: MeterType) {
        val rate = if (type == MeterType.PLN_ELECTRICITY) 1444.70 else 4200.0
        val defaultId = if (type == MeterType.PLN_ELECTRICITY) "541100234182" else "09234182"
        _uiState.value = _uiState.value.copy(
            selectedType = type,
            tariffRatePerUnit = rate,
            customerId = defaultId
        )
    }

    fun onOCRResultExtracted(idPel: String, standDigits: String) {
        val currentVal = standDigits.toDoubleOrNull() ?: _uiState.value.previousReading
        val prevVal = _uiState.value.previousReading
        val delta = (currentVal - prevVal).coerceAtLeast(0.0)

        _uiState.value = _uiState.value.copy(
            customerId = idPel.ifEmpty { _uiState.value.customerId },
            detectedDigits = standDigits.ifEmpty { _uiState.value.detectedDigits },
            usageDelta = delta,
            estimatedBill = delta * _uiState.value.tariffRatePerUnit
        )
    }
}
